// ROS2 Publisher Node — Pet Robot Sensor
#include "rclcpp/rclcpp.hpp"
#include "sensor_msgs/msg/laser_scan.hpp"

class PetMonitorNode : public rclcpp::Node {
public:
  PetMonitorNode() : Node("pet_monitor") {
    publisher_ = create_publisher<sensor_msgs::msg::LaserScan>(
      "/scan", rclcpp::QoS(10));
    timer_ = create_wall_timer(
      std::chrono::milliseconds(100),
      [this]() { publishScan(); });
  }
private:
  void publishScan() {
    auto msg = sensor_msgs::msg::LaserScan();
    msg.header.frame_id = "base_scan";
    msg.angle_min = -M_PI; msg.angle_max = M_PI;
    msg.range_max = 10.0f;
    publisher_->publish(msg);
  }
  rclcpp::Publisher<sensor_msgs::msg::LaserScan>::SharedPtr publisher_;
  rclcpp::TimerBase::SharedPtr timer_;
};

// Sensor Fusion — EKF State Prediction
struct VehicleState {
  Eigen::VectorXd x;  // [px, py, vx, vy, yaw]
  Eigen::MatrixXd P;  // Covariance matrix
};
VehicleState predict(VehicleState s, double dt) {
  Eigen::MatrixXd F = buildTransitionMatrix(dt);
  Eigen::MatrixXd Q = processNoise(dt);
  s.x = F * s.x;
  s.P = F * s.P * F.transpose() + Q;
  return s;
}
VehicleState updateGPS(VehicleState s, Eigen::Vector2d z) {
  Eigen::MatrixXd H = measurementMatrix();
  Eigen::MatrixXd R = gpsNoise();
  Eigen::VectorXd y = z - H * s.x;
  Eigen::MatrixXd S = H * s.P * H.transpose() + R;
  Eigen::MatrixXd K = s.P * H.transpose() * S.inverse();
  s.x = s.x + K * y;
  s.P = (Eigen::MatrixXd::Identity(5,5) - K*H) * s.P;
  return s;
}
